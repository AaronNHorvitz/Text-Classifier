void find(char* keyword) {
    struct dirent *entry;
    DIR *dir = opendir("/path/to/files");
    if (dir == NULL)
        return;

    while ((entry = readdir(dir)) != NULL) {
        if (strstr(entry->d_name, keyword) != NULL) {
            printf("%s\n", entry->d_name);
        }
    }
}
