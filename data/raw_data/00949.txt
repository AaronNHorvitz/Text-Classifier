import sqlite3
import requests

conn = sqlite3.connect('supplier_inventory.db')
cursor = conn.cursor()

def fetch_component_parts():
    response = requests.get('https://api.supplier.com/components')
    if response.status_code == 200:
        data = response.json()
        for component in data['components']:
            cursor.execute("INSERT INTO components (part_number, quantity) VALUES (?, ?)", (component['part_number'], component['quantity']))
    conn.commit()
