int find(int target) {
    int low = 0;
    int high = array_size - 1;

    while (high >= low) {
        int mid = (low + high) / 2;
        if (array[mid] == target)
            return mid;
        else if (array[mid] < target)
            low = mid + 1;
        else
            high = mid - 1;
    }
    return -1; // not found
}
